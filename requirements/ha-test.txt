# Test requirements for the Home Assistant custom integration.
#
# The versions are pinned so that the test run is reproducible.  The
# Home Assistant version is the oldest one the integration supports.
#
# pycares is held below 5.0: version 5 starts a background thread that
# survives the event loop shutdown, which the Home Assistant test harness
# reports as a leaked thread.
#
# acme and josepy are pinned to the pair that Home Assistant 2024.11 was
# released with.  Newer josepy releases drop the API that acme uses, which
# breaks the import of hass_nabucasa.
#
# The Home Assistant version is pinned as well.  The test harness of
# pytest-homeassistant-custom-component tracks a beta release of Home
# Assistant, and the beta behaves differently from the final release.
pytest-homeassistant-custom-component==0.13.375
homeassistant==2026.11.0
pytest-asyncio==1.4.1.post1
pytest-aiohttp==1.1.1
pytest-cov==7.1.0
pytest-socket==0.8.1
respx==0.23.1
pycares==4.11.0
acme==5.9.0
josepy==2.2.0
